import asyncio,json,sys
from pathlib import Path
from playwright.async_api import async_playwright
R=Path('/dev-server/.lovable/reports/canonical-heading-brief4')
async def main():
 out={};phase=sys.argv[1]
 async with async_playwright() as p:
  b=await p.chromium.launch(headless=True)
  for w in (390,320):
   c=await b.new_context(viewport={'width':w,'height':1800})
   for s in ('courses','list','about','score','single','few','you','picker','standout','rare','race','ranking','wagr','boards','achievement','explorecard','watchshelf','watchlead','featured'):
    page=await c.new_page();errors=[];
    if phase=='before':
     mapping=json.loads(Path('/tmp/browser/brief4/before-map.json').read_text())
     async def previous(route):
      from urllib.parse import urlsplit
      u=urlsplit(route.request.url)
      if u.path in mapping:
       response=await route.fetch(url='http://localhost:8080'+mapping[u.path]+('?' +u.query if u.query else ''));await route.fulfill(response=response)
      else: await route.continue_()
     await page.route('http://localhost:8080/src/**',previous)
    page.on('pageerror',lambda e:errors.append(str(e)))
    await page.goto('http://localhost:8080/src/__brief4_capture/index.html?screen='+s,wait_until='domcontentloaded');await page.wait_for_timeout(900)
    data=await page.evaluate('''() => ({text:document.getElementById('fixture')?.innerText||'',figures:[...document.querySelectorAll('#fixture *')].filter(e=>e.childElementCount===0 && /[0-9]/.test(e.textContent||'')).map(e=>{let r=document.createRange();r.selectNodeContents(e);let b=r.getBoundingClientRect();return {text:e.textContent,width:b.width,height:b.height,weight:getComputedStyle(e).fontWeight,column:e.getBoundingClientRect().width,overflow:e.scrollWidth>e.clientWidth}}),pageOverflow:document.documentElement.scrollWidth>innerWidth})''')
    data['errors']=errors;out[f'{s}-{w}']=data
    await page.screenshot(path=str(R/'captures'/phase/f'{s}-{w}.png'));print(s,w,'text',len(data['text']),'errors',errors[:2]);await page.close()
   await c.close()
  await b.close()
 (R/(phase+'-figures.json')).write_text(json.dumps(out,indent=2))
asyncio.run(main())
