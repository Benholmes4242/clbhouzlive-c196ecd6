import asyncio,json,sys
from pathlib import Path
from playwright.async_api import async_playwright
R=Path('/dev-server/.lovable/reports/canonical-heading-brief5')
async def main():
 phase=sys.argv[1];out={}
 async with async_playwright() as p:
  b=await p.chromium.launch(headless=True)
  for w in (390,320):
   c=await b.new_context(viewport={'width':w,'height':1800})
   for screen in ('empty','week','followers','insights','sample','holes','plays','tees','teeSheet','cardAgainst','behaviour','analyticsHealth','analyticsLive','journeys','screens','healthApp','healthGolf'):
    page=await c.new_page();errors=[];page.on('pageerror',lambda e:errors.append(str(e)))
    if phase=='before':
     mapping=json.loads(Path('/tmp/browser/brief5/before-map.json').read_text())
     async def previous(route):
      from urllib.parse import urlsplit
      u=urlsplit(route.request.url)
      if u.path in mapping:
       response=await route.fetch(url='http://localhost:8080'+mapping[u.path]+('?' +u.query if u.query else ''));await route.fulfill(response=response)
      else:await route.continue_()
     await page.route('http://localhost:8080/src/**',previous)
    await page.goto('http://localhost:8080/src/__brief5_capture/index.html?screen='+screen,wait_until='domcontentloaded');await page.wait_for_timeout(900)
    data=await page.evaluate('''() => ({text:document.querySelector('#fixture')?.innerText||'',figures:[...document.querySelectorAll('#fixture *')].filter(e=>e.childElementCount===0 && e.textContent?.trim()).map(e=>{const r=document.createRange();r.selectNodeContents(e);const b=r.getBoundingClientRect(),s=getComputedStyle(e);return {text:e.textContent,width:b.width,height:b.height,column:e.getBoundingClientRect().width,weight:s.fontWeight,fontFamily:s.fontFamily,selector:e.className||e.getAttributeNames().filter(n=>n.startsWith('data-')).join(','),overflow:e.scrollWidth>e.clientWidth}}),pageOverflow:document.documentElement.scrollWidth>innerWidth})''')
    data['errors']=errors;out[f'{screen}-{w}']=data
    await page.screenshot(path=str(R/'captures'/phase/f'{screen}-{w}.png'));print(screen,w,'text',len(data['text']),'errors',errors[:1]);await page.close()
   await c.close()
  await b.close()
 (R/(phase+'-figures.json')).write_text(json.dumps(out,indent=2))
asyncio.run(main())
